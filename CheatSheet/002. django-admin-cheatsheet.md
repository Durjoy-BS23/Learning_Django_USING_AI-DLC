# Django Admin Cheatsheet

## 🔵 INCEPTION PHASE - Setup & Configuration

### Create Superuser
```bash
python manage.py createsuperuser
# Prompts for: Username, Email, Password
```

### Access Admin Panel
```bash
python manage.py runserver
```

## 🟢 CONSTRUCTION PHASE - Model Registration & Customization

### Basic Model Registration
```python
# admin.py
from django.contrib import admin
from .models import Post

admin.site.register(Post)
```

### Model with __str__ Method
```python
# models.py
from django.db import models

class Post(models.Model):
    post_title = models.CharField(max_length=200)
    post_content = models.TextField()
    published_date = models.DateTimeField(auto_now=True)

    def __str__(self):
        return self.post_title
```

### Basic ModelAdmin Class
```python
# admin.py
from django.contrib import admin
from .models import Post

class PostAdmin(admin.ModelAdmin):
    pass

admin.site.register(Post, PostAdmin)
```

### Decorator Registration Pattern
```python
# admin.py
from django.contrib import admin

class PostAdmin(admin.ModelAdmin):
```

### Complete ModelAdmin Example
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    # Display fields in list view
    list_display = ['id', 'post_title', 'published_date', 'is_published']
    
    # Make fields clickable
    list_display_links = ['id', 'post_title']
    
    # Add sidebar filters
    list_filter = ['published_date', 'is_published']
    
    # Add search functionality
    search_fields = ['post_title', 'post_content']
    
    # Default ordering
    ordering = ['-published_date']
    
    # Items per page
    list_per_page = 25
```

### Custom Display Method
```python
from django.contrib import admin
from django.utils import timezone
import datetime

@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    list_display = ['title', 'published_date', 'is_recent']
    
    @admin.display(boolean=True, description='Published recently?')
    def is_recent(self, obj):
        seven_days_ago = timezone.now() - datetime.timedelta(days=7)
        return obj.published_date >= seven_days_ago
```

### Read-Only Fields
```python
@admin.register(Post)

    readonly_fields = ['created_at', 'updated_at']
```

### Field Ordering in Forms
```python
@admin.register(Post)

    fields = ['title', 'content', 'category', 'status']
```

### Exclude Fields from Forms
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    exclude = ['created_at', 'updated_at']
```

### Date Hierarchy
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    date_hierarchy = 'published_date'
```

### Custom Admin Actions
```python
@admin.action(description='Mark as published')
def make_published(modeladmin, request, queryset):
    queryset.update(is_published=True)

@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    actions = [make_published]
```

### Query Optimization
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    list_display = ['title', 'author', 'category']
    
    def get_queryset(self, request):
        return super().get_queryset(request).select_related('author', 'category')
```

### Save Model Hook
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    def save_model(self, request, obj, form, change):
        obj.author = request.user
        super().save_model(request, obj, form, change)
```

## search_fields Limitations

### Works With
- CharField
- TextField
- Related fields with `__` lookup (e.g., `author__name`)

### Does NOT Work With
- IntegerField
- DateField
- BooleanField
- ForeignKey directly (use `__` lookup)

## Common Patterns

### Register Multiple Models
```python
from django.contrib import admin
from .models import Post, Category, Author

@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    list_display = ['title', 'published_date']

@admin.register(Category)
class CategoryAdmin(admin.ModelAdmin):
    list_display = ['name']

@admin.register(Author)
class AuthorAdmin(admin.ModelAdmin):
    list_display = ['name', 'email']
```

### Custom Admin Site
```python
# admin.py
from django.contrib.admin import AdminSite

class MyAdminSite(AdminSite):
    site_header = 'My Custom Admin'
    site_title = 'My Admin Portal'
    index_title = 'Welcome to My Admin'

my_admin_site = MyAdminSite(name='myadmin')

# urls.py
from django.urls import path
from .admin import my_admin_site

urlpatterns = [
    path('my-admin/', my_admin_site.urls),
]
```

### Soft Delete Pattern
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    def get_queryset(self, request):
        return super().get_queryset(request).filter(is_deleted=False)
```

## Common Pitfalls

### Wrong
```python
# Forgetting to import model
admin.site.register(Post)  # NameError

# Wrong field names in list_display
list_display = ['title']  # Model has post_title

# Searching non-text fields
search_fields = ['id', 'price']  # Won't work

# Not restarting server after registration changes
# Causes "already registered" error

# Overloading list_display
list_display = ['id', 'title', 'content', 'author', 'category', 'status', 'created_at', 'updated_at']
```

### Right
```python
# Import model first
from .models import Post
admin.site.register(Post)

# Use exact field names
list_display = ['post_title']

# Use filters for non-text fields
search_fields = ['title']
list_filter = ['price', 'created_at']

# Restart server when switching registration methods

# Focused list_display
list_display = ['id', 'title', 'status', 'created_at']
```

### __str__ Best Practices
```python
# Good - Returns identifying information
def __str__(self):
    return self.title

# Handle None values
def __str__(self):
    return self.title or f"Untitled #{self.id}"

# Multiple fields
def __str__(self):
    return f"{self.title} ({self.published_date})"
```

### ModelAdmin Naming Convention
```python

class PostAdmin(admin.ModelAdmin):
    pass

class UserAdmin(admin.ModelAdmin):
    pass
```

## Performance Tips

### Optimize Queries
```python
def get_queryset(self, request):
    return super().get_queryset(request).select_related('author', 'category')
```

### Database Indexes
```python
class Post(models.Model):
    title = models.CharField(max_length=200, db_index=True)
    published_date = models.DateTimeField(db_index=True)
```

### Limit search_fields
```python
# Good - Essential fields only
search_fields = ['title']

# Bad - Too many fields (slow on large datasets)
search_fields = ['title', 'content', 'summary', 'notes', 'description']
```

### Custom Branding Template
